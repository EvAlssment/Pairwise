cmake_minimum_required(VERSION 3.25)
project(PairwiseWindows VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
include(CTest)
option(PAIRWISE_BUILD_APP "Build native desktop/media client" ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Gui Widgets Test)
add_library(pairwise_core src/Protocol.cpp src/Api.cpp src/CallSession.cpp)
target_include_directories(pairwise_core PUBLIC src)
target_link_libraries(pairwise_core PUBLIC Qt6::Core Qt6::Network Qt6::Gui)
if(MSVC)
  target_compile_options(pairwise_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(pairwise_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PAIRWISE_BUILD_APP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Multimedia)
  find_package(LiveKit 1.13.1 EXACT CONFIG REQUIRED)
  qt_add_executable(Pairwise WIN32 src/main.cpp src/MainWindow.cpp src/Media.cpp
    src/Capture.cpp src/Collaboration.cpp src/VideoCanvas.cpp src/Platform.cpp)
  target_link_libraries(Pairwise PRIVATE pairwise_core Qt6::Widgets Qt6::Multimedia LiveKit::livekit)
  if(BUILD_TESTING)
    qt_add_executable(pairwise_media_test tests/MediaIntegration.cpp)
    target_link_libraries(pairwise_media_test PRIVATE pairwise_core LiveKit::livekit)
  endif()
  target_compile_definitions(Pairwise PRIVATE PAIRWISE_VERSION="${PROJECT_VERSION}")
  if(WIN32)
    target_compile_definitions(Pairwise PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    target_link_libraries(Pairwise PRIVATE Advapi32 User32)
    set_property(TARGET Pairwise PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    install(TARGETS Pairwise RUNTIME DESTINATION .)
    # Include LiveKit's transitive DLLs; windeployqt supplies Qt/FFmpeg/MSVC.
    get_target_property(livekit_dll LiveKit::livekit IMPORTED_LOCATION_RELEASE)
    if(NOT livekit_dll)
      get_target_property(livekit_dll LiveKit::livekit IMPORTED_LOCATION)
    endif()
    get_filename_component(livekit_bin "${livekit_dll}" DIRECTORY)
    install(DIRECTORY "${livekit_bin}/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
  endif()
endif()
if(BUILD_TESTING)
  if(PAIRWISE_BUILD_APP)
    qt_add_executable(pairwise_capture_tests tests/CaptureTests.cpp src/Capture.cpp src/Platform.cpp)
    target_link_libraries(pairwise_capture_tests PRIVATE pairwise_core Qt6::Widgets Qt6::Multimedia Qt6::Test)
    add_test(NAME windows-capture COMMAND pairwise_capture_tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/windows-capture.xml,junitxml")
    set_tests_properties(windows-capture PROPERTIES TIMEOUT 45)
    if(WIN32)
      target_compile_definitions(pairwise_capture_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
      target_link_libraries(pairwise_capture_tests PRIVATE Advapi32 User32)
    else()
      set_tests_properties(windows-capture PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    qt_add_executable(pairwise_audio_tests tests/AudioTests.cpp src/Media.cpp)
    target_link_libraries(pairwise_audio_tests PRIVATE pairwise_core Qt6::Test LiveKit::livekit)
    add_test(NAME windows-audio COMMAND pairwise_audio_tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/windows-audio.xml,junitxml")
    set_tests_properties(windows-audio PROPERTIES TIMEOUT 30)
    if(WIN32)
      set_tests_properties(windows-audio PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${livekit_bin}")
    endif()
  endif()
  qt_add_executable(pairwise_viewer_tests tests/VideoCanvasTests.cpp src/VideoCanvas.cpp)
  target_link_libraries(pairwise_viewer_tests PRIVATE pairwise_core Qt6::Widgets Qt6::Test)
  add_test(NAME windows-viewer COMMAND pairwise_viewer_tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/windows-viewer.xml,junitxml")
  set_tests_properties(windows-viewer PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
  qt_add_executable(pairwise_tests tests/CoreTests.cpp)
  target_link_libraries(pairwise_tests PRIVATE pairwise_core Qt6::Test)
  target_compile_definitions(pairwise_tests PRIVATE PAIRWISE_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../shared/protocol-fixtures")
  add_test(NAME windows-core COMMAND pairwise_tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/windows-core.xml,junitxml")
  set_tests_properties(windows-core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32)
    qt_add_executable(pairwise_platform_tests tests/WindowsPlatformTests.cpp src/Platform.cpp)
    target_link_libraries(pairwise_platform_tests PRIVATE pairwise_core Qt6::Widgets Qt6::Test Advapi32 User32)
    target_compile_definitions(pairwise_platform_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
    add_test(NAME windows-platform COMMAND pairwise_platform_tests -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/windows-platform.xml,junitxml")
    set_tests_properties(windows-platform PROPERTIES TIMEOUT 30)
  endif()
endif()
